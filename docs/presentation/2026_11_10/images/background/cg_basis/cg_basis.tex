\documentclass[10pt,border=2pt,multi=tikzpicture]{standalone}

\makeatletter\def\input@path{{../../}{../../../}}\makeatother % figpreamble.sty (images/) et diapocolors.sty (racine)

\usepackage{figpreamble}

% une couleur par fonction de base (ni bleu ni vert, réservés aux hybridations ; ni rouge ni orange, réservés aux points clés du texte)
\definecolor{bas1}{RGB}{128,60,170}
\definecolor{bas2}{RGB}{160,90,40}
\definecolor{bas3}{RGB}{210,50,150}
\definecolor{bas4}{RGB}{120,120,0}

\newcommand{\frameandbasis}{
	\useasboundingbox (-0.2,-0.15) rectangle (3.9,1.6);

	% maillage
	\draw[neutral] (-0.1,0) -- (3.1,0);
	\foreach \x in {0,1,2,3} { \draw[neutral, thick] (\x,0) -- (\x,-0.06); }

	% chapeaux phi_i : un par noeud, chacun sa couleur, étiqueté au sommet
	\draw[bas1, thick] (0,1) -- (1,0);
	\draw[bas2, thick] (0,0) -- (1,1) -- (2,0);
	\draw[bas3, thick] (1,0) -- (2,1) -- (3,0);
	\draw[bas4, thick] (2,0) -- (3,1);
	\foreach \x/\i in {0/1, 1/2, 2/3, 3/4} { \node[font=\footnotesize, bas\i, anchor=south] at (\x,0.98) {$\phi_\i$}; }

	% titre, au-dessus (comme avant)
	\node[font=\footnotesize, neutral, anchor=south west] at (-0.1,1.3) {$N_\text{cells}=3$, $N_\text{dofs}=4$ ($k=1$)};
}

% dofs beta_i : un par noeud, de la couleur de sa fonction de base, juste sous le maillage
\newcommand{\globaldofs}{
	\foreach \x/\i in {0/1, 1/2, 2/3, 3/4} { \node[bas\i, font=\footnotesize] at (\x,-0.2) {$\beta_\i$}; }
}

% u_h continue = somme des beta_i phi_i
\newcommand{\continuouspart}{
	\draw[black, very thick] (0,0.3) -- (1,0.7) -- (2,0.4) -- (3,0.85);
	\node[font=\footnotesize, anchor=west] at (3.15,0.85) {$u_h$};
}

\begin{tikzpicture}[x=0.9cm,y=1.7cm]
	\frameandbasis
	% \globaldofs
\end{tikzpicture}

	% \globaldofs
