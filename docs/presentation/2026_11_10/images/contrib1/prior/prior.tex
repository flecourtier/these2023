\documentclass[10pt,border=2pt,multi=tikzpicture]{standalone}

\usepackage{figpreamble}
\usepackage{pgfplots}
\usepgfplotslibrary{groupplots}
\pgfplotsset{compat=newest}

\newif\ifzoom
% #1 = 0 : sans zoom (le deuxième axe est conservé mais masqué, pour garder le même cadre) ; #1 = 1 : avec zoom
\newcommand{\priorfig}[1]{%
	\ifnum#1=1 \zoomtrue\else\zoomfalse\fi
	\begin{tikzpicture}
		\begin{groupplot}[group style={group size=1 by 2, vertical sep=0.7em, group name=mygroupplot},
				enlargelimits=false, height=3.0cm, width=3.25cm, scale only axis, clip mode=individual,
				tick label style={font=\scriptsize}, legend style={draw=none}, legend columns=1, legend pos=north east]
			\nextgroupplot[ymin=-0.55, ymax=0.55, xmin=0, xmax=1, ytick={-0.6,0.6}, xtick={-0.1,1.1}]
			\addplot[color=orange, line width=1.5pt] table[x=x, y=u] {derivatives_mu1_PINN.txt}; \addlegendentry{$u$}
			\addplot[color=blue, line width=1pt, densely dotted] table[x=x, y=u_theta] {derivatives_mu1_PINN.txt}; \addlegendentry{$u_\theta$}
			\coordinate (c1) at (axis cs:0.55,-0.15);
			\coordinate (c2) at (axis cs:0.75,-0.15);
			\ifzoom\draw (c1) rectangle (axis cs:0.75,0.05);\fi

			\ifzoom
				\nextgroupplot[ymin=-0.15, ymax=0.05, xmin=0.55, xmax=0.75, ytick={-0.6,0.6}, xtick={-0.1,1.1}, name=ax2, height=2.5cm, width=2.5cm]
				\addplot[color=orange, line width=1.5pt] table[x=x, y=u] {derivatives_mu1_PINN.txt};
				\addplot[color=blue, line width=1pt, densely dotted] table[x=x, y=u_theta] {derivatives_mu1_PINN.txt};
				\draw[thick, darkred, <->] (axis cs:0.65,-0.08) -- (axis cs:0.65,-0.042) node[midway, left] {$p_h^+$};
			\else
				\nextgroupplot[ymin=-0.15, ymax=0.05, xmin=0.55, xmax=0.75, ytick={-0.6,0.6}, xtick={-0.1,1.1}, name=ax2, height=2.5cm, width=2.5cm, hide axis]
			\fi
		\end{groupplot}
		\ifzoom
			\draw[dashed] (c1) -- (ax2.north west);
			\draw[dashed] (c2) -- (ax2.north east);
		\fi
	\end{tikzpicture}}
