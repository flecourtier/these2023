%% Figure standalone : principe d'une méthode numérique, en bandeau horizontal (encodage -> approximation -> décodage).
%% Les trois étapes sont visibles dès la page 1 ; ce qui n'est pas encore atteint est estompé.
%% Boîtes « function » et « vector » (toutes à fond blanc, le texte dit la nature de l'inconnue) : on quitte l'espace des fonctions, on résout, on y revient.
%% Page 1 : encodage (rouge). Page 2 : approximation (rouge). Page 3 : décodage (rouge). Les étapes passées sont noires. Page 4 : tout en noir, rien en rouge ni estompé.
%% Même cadre sur les trois pages. Figure commune : neutre.
%% Compilation : à l'enregistrement dans VS Code (LaTeX Workshop, commentaire magique ci-dessous),
%% ou images/build_fig.sh images/background/encdec/encdec.tex
% !TEX program = lualatex
\documentclass[10pt,border=2pt,multi=tikzpicture]{standalone}

\makeatletter\def\input@path{{../../}{../../../}}\makeatother % figpreamble.sty (images/) et diapocolors.sty (racine)

\usepackage{figpreamble}

\colorlet{faded}{neutral!30}

\tikzset{
	box/.style={draw=black, thick, rounded corners=3pt, minimum height=0.95cm, align=center, font=\footnotesize},
	fn/.style={box, fill=white},        % espace de fonctions (dimension infinie)
	vec/.style={box, fill=white},       % espace de vecteurs (dimension finie)
	arr/.style={-{Stealth[length=2.4mm]}, very thick},
	steplabel/.style={font=\footnotesize\bfseries, anchor=south, align=center}
}

% #1, #2, #3 : couleur de l'encodage, de l'approximation, du décodage (red = en cours, black = fait, faded = à venir)
% #4, #5 : couleur des deux dernières boîtes (black ou faded)
\newcommand{\encdec}[5]{
	\useasboundingbox (-0.1,-0.6) rectangle (14.0,1.25);

	\node[fn, minimum width=3.0cm] (pde) at (1.5,0) {PDE\\(unknown: function)};
	\node[vec, minimum width=3.0cm] (disc) at (5.4,0) {discrete problem\\(unknown: vector)};
	\node[vec, minimum width=2.6cm, draw=#4, text=#4] (sol) at (9.1,0) {discrete solution\\(solve)};
	\node[fn, minimum width=2.6cm, draw=#5, text=#5] (uh) at (12.6,0) {approx. solution\\(function)};

	\draw[arr, #1] (pde.east) -- (disc.west);
	\node[steplabel, #1] at (3.45,0.52) {Encoding};
	\node[#1] at (3.45,-0.37) {\stepcirc{1}};
	\draw[arr, #2] (disc.east) -- (sol.west);
	\node[steplabel, #2] at (7.35,0.52) {Approximation};
	\node[#2] at (7.35,-0.37) {\stepcirc{2}};
	\draw[arr, #3] (sol.east) -- (uh.west);
	\node[steplabel, #3] at (10.85,0.52) {Decoding};
	\node[#3] at (10.85,-0.37) {\stepcirc{3}};
}

\begin{document}

\begin{tikzpicture}[x=1cm,y=1cm]
	\encdec{red}{faded}{faded}{faded}{faded}
\end{tikzpicture}

\begin{tikzpicture}[x=1cm,y=1cm]
	\encdec{black}{red}{faded}{black}{faded}
\end{tikzpicture}

\begin{tikzpicture}[x=1cm,y=1cm]
	\encdec{black}{black}{red}{black}{black}
\end{tikzpicture}

\begin{tikzpicture}[x=1cm,y=1cm]
	\encdec{black}{black}{black}{black}{black}
\end{tikzpicture}

\end{document}
